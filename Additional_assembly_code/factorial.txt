      noop             # for store data
      noop             # for store data
      lw 0 1 input     # get input to x1
      lw 0 6 one       # x6 = 1
      lw 0 7 one       # x7 = 1
fact  lw 0 3 stop      # x3 = 1 or base case
      beq 3 1 end      # go to end of program if (input == base case)   
      sw 0 1 0         # store data
      sw 6 7 0         # store data
      lw 0 4 minus     # x4 = -1 (useing for decreate value)
      add 1 4 1        # decreate value in x1
mult  lw 0 2 0         # load data from address 0 
      lw 0 7 zero      # set x7 to identity addition and this reg carry result
      lw 0 3 1         # load data from address 1
      lw 0 4 one       # x4 = 1 use this to get 0 or 1 at current index
loop  beq 4 0 fact     # go back to fact if (value in x4 == 0)
      nand 2 4 5       # get current bit
      nand 5 5 5       # not nand
      beq 5 0 is0      # skip next line if current interesting bit is 0
      add 3 7 7        # from flowchart
is0   add 4 4 4        # shift right (not the real shift right)
      add 3 3 3        # shift left
      beq 0 0 loop     # go back to loop

end   halt 

one  .fill 1
stop .fill 1
minus .fill -1
zero .fill 0
input .fill 10
